cmake_minimum_required(VERSION 3.18)
project(synthkb CXX C)
set(CMAKE_CXX_STANDARD 17)
set(GLUE ${ANDROID_NDK}/sources/android/native_app_glue)
add_library(native_app_glue STATIC ${GLUE}/android_native_app_glue.c)
target_include_directories(native_app_glue PUBLIC ${GLUE})
add_library(synthkb SHARED main.cpp app.cpp engine.cpp)
target_include_directories(synthkb PRIVATE ${GLUE})
target_compile_options(synthkb PRIVATE -O3 -ffast-math -Wall)
target_link_options(synthkb PRIVATE -u ANativeActivity_onCreate)
target_link_libraries(synthkb native_app_glue android log aaudio)
